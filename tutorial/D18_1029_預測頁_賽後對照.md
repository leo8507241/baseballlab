# ---------- Day 18：季後賽 ----------
def find_season_id(year, keyword):
    """在中職所有賽季中，找出標題含有 keyword（例如「台灣大賽」）的賽季代碼；找不到回傳 None。"""
    seasons = get_json("/api/leagues/CPBL")["data"]["seasons"]
    for s in seasons:
        if s.get("year") == year and keyword in s.get("title", ""):
            return s["uniqid"]
    return None
```

# ---------- Day 18：賽後對照 ----------
st.header("賽後對照：模型猜對了幾場？")

@st.cache_data(ttl=3600)  # 一小時內重複開頁不會再去抓
def postseason_games():
    season_id = fetch.find_season_id(2026, config.POSTSEASON_TITLE)
    return fetch.fetch_games(season_id, refresh=True) if season_id else None

games = postseason_games()
if games is None or games.empty:
    st.info("台灣大賽還沒開打，或資料還沒更新。比賽結束後重新整理就會出現對照表。")
else:
    games = games.sort_values("date").reset_index(drop=True)
    rows = []
    for i, gm in games.iterrows():
        a_home = gm["home"] == A_NAME
        p = model.game_prob(A["pythag"], B["pythag"], a_home, config.HOME_ADVANTAGE)
        fav = A_NAME if p >= 0.5 else B_NAME
        winner = gm["home"] if gm["winner_side"] == "HOME" else gm["away"]
        rows.append({"場次": f"G{i + 1}", "日期": gm["date"], "主場": gm["home"],
                     "模型看好": fav, "預測勝率": f"{max(p, 1 - p):.0%}",
                     "比分": f"{gm['away']} {gm['away_runs']} : {gm['home_runs']} {gm['home']}",
                     "實際勝方": winner, "猜對": "✓" if fav == winner else "✗"})
    table = pd.DataFrame(rows)
    hit = (table["猜對"] == "✓").mean()
    ui.kpi("模型命中率", f"{hit:.0%}", f"{len(table)} 場中猜對 {(table['猜對'] == '✓').sum()} 場")
    st.dataframe(table, hide_index=True, width="stretch")
```
